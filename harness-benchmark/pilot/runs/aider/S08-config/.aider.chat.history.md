﻿function Read-KeyValue {
    param([string]$Text)
    $h = @{}
    foreach ($line in ($Text -split "`n")) {
        $l = $line.Trim()
        if ($l -eq '' -or $l.StartsWith('#')) { continue }
        $idx = $l.IndexOf('=')
        if ($idx -lt 0) { continue }
        $key = $l.Substring(0, $idx).Trim()

}
