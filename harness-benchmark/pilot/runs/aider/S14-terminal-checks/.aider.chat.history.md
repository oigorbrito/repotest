tools\Run-Checks.ps1
```
﻿$base = Split-Path -Parent $PSScriptRoot
$pastaChecks = Join-Path $base 'checks'
$pastaRelatorio = Join-Path $base 'relatorio'
$arqResultado = Join-Path $pastaRelatorio 'resultados.txt'

if (-not (Test-Path $pastaRelatorio)) {
    New-Item -ItemType Directory -Path $pastaRelatorio | Out-Null
}

$linhas = @()

$scripts = Get-ChildItem -Path $pastaChecks -Filter '*.ps1' -File | Sort-Object Name

foreach ($script in $scripts) {
    $nome = [System.IO.Path]::GetFileNameWithoutExtension($script.Name)
    $processo = Start-Process -FilePath 'powershell.exe' `
        -ArgumentList @('-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', $script.FullName) `
        -Wait -PassThru -NoNewWindow
    if ($processo.ExitCode -eq 0) {
        $linhas += "OK $nome"
    } else {
        $linhas += "FALHOU $nome"
    }
}

[System.IO.File]::WriteAllText($arqResultado, (($linhas -join "`n") + "`n"), (New-Object System.Text.UTF8Encoding($false)))
```
